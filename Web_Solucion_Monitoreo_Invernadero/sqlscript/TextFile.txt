DROP DATABASE IF EXISTS monitoreo_orquideas;
CREATE DATABASE monitoreo_orquideas CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE monitoreo_orquideas;

-- 1. USUARIOS DEL SISTEMA
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    email VARCHAR(120) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    rol VARCHAR(30) NOT NULL DEFAULT 'Operador',
    state TINYINT(1) NOT NULL DEFAULT 1
);

-- 2. PLANTAS (Cada especie tiene sus 4 umbrales y registra qué usuario la ingresó)
CREATE TABLE plantas (
    id_planta INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion VARCHAR(255) NULL,
    temperatura_minima DECIMAL(5,2) NOT NULL,
    temperatura_maxima DECIMAL(5,2) NOT NULL,
    humedad_minima DECIMAL(5,2) NOT NULL,
    humedad_maxima DECIMAL(5,2) NOT NULL,
    id_usuario_creacion INT NOT NULL,
    fecha_actualizacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    state TINYINT(1) NOT NULL DEFAULT 1,
    CONSTRAINT fk_planta_usuario FOREIGN KEY (id_usuario_creacion) REFERENCES usuarios(id_usuario)
);

-- 3. INVERNADEROS (3 sectores; relación estricta 1:1 con planta mediante UNIQUE)
CREATE TABLE invernaderos (
    id_invernadero INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    ubicacion VARCHAR(150) NOT NULL,
    id_planta INT NOT NULL UNIQUE,
    frecuencia_registro_min INT NOT NULL DEFAULT 60,
    id_usuario_modificacion INT NOT NULL,
    fecha_modificacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    state TINYINT(1) NOT NULL DEFAULT 1,
    CONSTRAINT fk_invernadero_planta FOREIGN KEY (id_planta) REFERENCES plantas(id_planta),
    CONSTRAINT fk_invernadero_usuario FOREIGN KEY (id_usuario_modificacion) REFERENCES usuarios(id_usuario)
);

-- 4. SENSORES / ARDUINOS ESP32 (1 dispositivo dedicado por cada invernadero = 3 en total)
CREATE TABLE sensores (
    id_sensor INT AUTO_INCREMENT PRIMARY KEY,
    id_invernadero INT UNIQUE,
    codigo_dispositivo VARCHAR(50) NOT NULL UNIQUE,
    nombre_nodo VARCHAR(100) NOT NULL,
    estado_conexion VARCHAR(30) NOT NULL DEFAULT 'Desconectado',
    ultima_senal DATETIME NULL,
    state TINYINT(1) NOT NULL DEFAULT 1,
    CONSTRAINT fk_sensor_invernadero FOREIGN KEY (id_invernadero) REFERENCES invernaderos(id_invernadero)
);

-- 5. CALENDARIO DE RIEGO (Por invernadero; registra el usuario que estableció el horario)
CREATE TABLE calendario_riego (
    id_calendario_riego INT AUTO_INCREMENT PRIMARY KEY,
    id_invernadero INT NOT NULL,
    id_usuario INT NOT NULL,
    fecha_especifica DATE NULL,
    dia_semana VARCHAR(20) NULL,
    hora_riego TIME NOT NULL,
    tiempo_riego_seg INT NOT NULL DEFAULT 10,
    pospuesto_hasta DATE NULL,
    fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    state TINYINT(1) NOT NULL DEFAULT 1,
    CONSTRAINT fk_calendario_invernadero FOREIGN KEY (id_invernadero) REFERENCES invernaderos(id_invernadero),
    CONSTRAINT fk_calendario_usuario FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);

-- 6. REGISTRO HISTÓRICO AMBIENTAL
CREATE TABLE registro_historico (
    id_registro INT AUTO_INCREMENT PRIMARY KEY,
    id_invernadero INT NOT NULL,
    humedad DECIMAL(5,2) NOT NULL,
    temperatura DECIMAL(5,2) NOT NULL,
    fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    state TINYINT(1) NOT NULL DEFAULT 1,
    CONSTRAINT fk_historico_invernadero FOREIGN KEY (id_invernadero) REFERENCES invernaderos(id_invernadero)
);

-- 7. BITÁCORA DE RIEGO Y ALERTAS AUTOMÁTICAS
CREATE TABLE bitacora_riego_alertas (
    id_bitacora INT AUTO_INCREMENT PRIMARY KEY,
    id_invernadero INT NOT NULL,
    tipo_evento VARCHAR(50) NOT NULL,
    descripcion VARCHAR(255) NOT NULL,
    tiempo_activacion_seg INT NOT NULL DEFAULT 0,
    humedad_momento DECIMAL(5,2) NOT NULL,
    temperatura_momento DECIMAL(5,2) NOT NULL,
    fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    leida TINYINT(1) NOT NULL DEFAULT 0,
    CONSTRAINT fk_bitacora_invernadero FOREIGN KEY (id_invernadero) REFERENCES invernaderos(id_invernadero)
);

-- 8. AUDITORÍA DE CAMBIOS MANUALES (Trazabilidad completa de acciones por usuario)
CREATE TABLE auditoria_cambios (
    id_auditoria INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    accion VARCHAR(80) NOT NULL,
    entidad_afectada VARCHAR(50) NOT NULL,
    detalle VARCHAR(255) NOT NULL,
    fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_auditoria_usuario FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
);
