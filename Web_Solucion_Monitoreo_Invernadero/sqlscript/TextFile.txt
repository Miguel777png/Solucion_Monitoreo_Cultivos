-- ============================================================================
-- BASE DE DATOS: Sistema IoT de Monitoreo y Riego de Orquídeas
-- Motor: MySQL 8.0+ (InnoDB / utf8mb4)
-- ============================================================================

CREATE DATABASE IF NOT EXISTS monitoreo_orquideas
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE monitoreo_orquideas;

-- ----------------------------------------------------------------------------
-- 1. TABLA: USUARIOS
-- ----------------------------------------------------------------------------
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(120) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    state BOOLEAN NOT NULL DEFAULT TRUE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 2. TABLA: PLANTAS
-- Almacena los umbrales óptimos y críticos de cada especie
-- ----------------------------------------------------------------------------
CREATE TABLE plantas (
    id_planta INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    temperatura_maxima DECIMAL(5,2) NOT NULL,
    temperatura_minima DECIMAL(5,2) NOT NULL,
    humedad_maxima DECIMAL(5,2) NOT NULL,
    humedad_minima DECIMAL(5,2) NOT NULL,
    state BOOLEAN NOT NULL DEFAULT TRUE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 3. TABLA: INVERNADEROS
-- Relación 1:1 con PLANTAS (garantizada mediante restricción UNIQUE en id_planta)
-- ----------------------------------------------------------------------------
CREATE TABLE invernaderos (
    id_invernadero INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_planta INT NOT NULL UNIQUE,
    ubicacion VARCHAR(150) NOT NULL,
    frecuencia_registro_min INT NOT NULL DEFAULT 60,
    state BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT fk_invernadero_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_invernadero_planta
        FOREIGN KEY (id_planta) REFERENCES plantas(id_planta)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 4. TABLA: SENSORES
-- Catálogo de dispositivos emparejados (Wi-Fi / Bluetooth) y estado de conexión
-- ----------------------------------------------------------------------------
CREATE TABLE sensores (
    id_sensor INT AUTO_INCREMENT PRIMARY KEY,
    id_invernadero INT NOT NULL,
    codigo_dispositivo VARCHAR(50) NOT NULL UNIQUE,
    tipo_conexion ENUM('WiFi', 'Bluetooth') NOT NULL DEFAULT 'WiFi',
    estado_conexion ENUM('Conectado', 'Desconectado') NOT NULL DEFAULT 'Desconectado',
    ultima_senal DATETIME NULL,
    state BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT fk_sensor_invernadero
        FOREIGN KEY (id_invernadero) REFERENCES invernaderos(id_invernadero)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 5. TABLA: REGISTRO_HISTORICO
-- Almacena únicamente las mediciones periódicas según frecuencia_registro_min
-- ----------------------------------------------------------------------------
CREATE TABLE registro_historico (
    id_registro INT AUTO_INCREMENT PRIMARY KEY,
    id_invernadero INT NOT NULL,
    humedad DECIMAL(5,2) NOT NULL,
    temperatura DECIMAL(5,2) NOT NULL,
    fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    state BOOLEAN NOT NULL DEFAULT TRUE,
    INDEX idx_historico_inv_fecha (id_invernadero, fecha_hora),
    CONSTRAINT fk_historico_invernadero
        FOREIGN KEY (id_invernadero) REFERENCES invernaderos(id_invernadero)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 6. TABLA: BITACORA_RIEGO_ALERTAS
-- Registra eventos asíncronos: riegos programados, emergencia, pospuestos y alertas
-- ----------------------------------------------------------------------------
CREATE TABLE bitacora_riego_alertas (
    id_evento INT AUTO_INCREMENT PRIMARY KEY,
    id_invernadero INT NOT NULL,
    tipo_evento ENUM(
        'RIEGO_PROGRAMADO',
        'RIEGO_EMERGENCIA',
        'RIEGO_POSPUESTO',
        'ALERTA_UMBRAL',
        'ERROR_SENSOR'
    ) NOT NULL,
    descripcion VARCHAR(255) NOT NULL,
    tiempo_activacion_seg INT NULL,
    humedad_momento DECIMAL(5,2) NULL,
    temperatura_momento DECIMAL(5,2) NULL,
    fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    leida BOOLEAN NOT NULL DEFAULT FALSE,
    INDEX idx_bitacora_inv_fecha (id_invernadero, fecha_hora),
    CONSTRAINT fk_bitacora_invernadero
        FOREIGN KEY (id_invernadero) REFERENCES invernaderos(id_invernadero)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ----------------------------------------------------------------------------
-- 7. TABLA: CALENDARIO_RIEGO
-- Permite asignar múltiples días o fechas específicas de riego por planta
-- ----------------------------------------------------------------------------
CREATE TABLE calendario_riego (
    id_calendario_riego INT AUTO_INCREMENT PRIMARY KEY,
    id_planta INT NOT NULL,
    fecha_especifica DATE NULL,
    dia_semana VARCHAR(15) NULL,
    hora_riego TIME NOT NULL,
    tiempo_riego_seg INT NOT NULL DEFAULT 15,
    pospuesto_hasta DATE NULL,
    state BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT fk_calendario_planta
        FOREIGN KEY (id_planta) REFERENCES plantas(id_planta)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;
